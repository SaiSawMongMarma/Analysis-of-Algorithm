
#include <stdio.h>
#include <string.h>

#define MAX 100

// Function Prototypes
void lcs_length(char *X, char *Y, int m, int n, int c[MAX][MAX], char b[MAX][MAX]);
void print_lcs(char b[MAX][MAX], char *X, int i, int j);

int main() {

    char X[] = "ABCBDAB";
    char Y[] = "BDCABA";

    int m = strlen(X);
    int n = strlen(Y);

    // c table stores lengths, b table stores tracking directions
    int c[MAX][MAX];
    char b[MAX][MAX];

    // Shift strings right by 1 index to align with 1-based indexing in slides
    char X_shifted[MAX] = " ";
    char Y_shifted[MAX] = " ";
    strcat(X_shifted, X);
    strcat(Y_shifted, Y);

    printf("Sequence X: %s\n", X);
    printf("Sequence Y: %s\n\n", Y);

    // Step 1: Run LCS-Length to populate the matrices
    lcs_length(X_shifted, Y_shifted, m, n, c, b);

    // Step 2: Output the results
    printf("Length of LCS: %d\n", c[m][n]);
    printf("Longest Common Subsequence: ");
    print_lcs(b, X_shifted, m, n);
    printf("\n");

    return 0;
}

// First Algorithm: Computes the dynamic programming matrices
void lcs_length(char *X, char *Y, int m, int n, int c[MAX][MAX], char b[MAX][MAX]) {
    // Base Case initialization 
    for (int i = 1; i <= m; i++) {
        c[i][0] = 0;
    }
    // Base Case initialization 
    for (int j = 0; j <= n; j++) {
        c[0][j] = 0;
    }

    // The DP Matrix Core loops 
    for (int i = 1; i <= m; i++) {
        for (int j = 1; j <= n; j++) {
            if (X[i] == Y[j]) {                      
                c[i][j] = c[i - 1][j - 1] + 1;      
                b[i][j] = 'D';                      //('D' = Diagonal ↖)
            }
            else if (c[i - 1][j] >= c[i][j - 1]) {  
                c[i][j] = c[i - 1][j];              
                b[i][j] = 'U';                      //('U' = Up ↑)
            }
            else {
                c[i][j] = c[i][j - 1];             
                b[i][j] = 'L';                      //('L' = Left ←)
            }
        }
    }
}

// Second Algorithm: Recursively backtracks and prints the string sequence
void print_lcs(char b[MAX][MAX], char *X, int i, int j) {
    // Base Case check (Lines 1-2)
    if (i == 0 || j == 0) {
        return;
    }

    // Tracing conditions (Lines 3-8)
    if (b[i][j] == 'D') {
        print_lcs(b, X, i - 1, j - 1);
        printf("%c", X[i]);            
    }
    else if (b[i][j] == 'U') {
        print_lcs(b, X, i - 1, j);     
    }
    else {
        print_lcs(b, X, i, j - 1);     
    }
}
