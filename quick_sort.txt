#include <stdio.h>

int exchange(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
    return 0;
}

// PARTITION(A, p, r)
int partition(int A[], int p, int r) {
    // 1 x = A[r]
    int x = A[r];

    // 2 i = p - 1
    int i = p - 1;

    // 3 for j = p to r - 1
    for (int j = p; j <= r - 1; j++) {
        // 4 if A[j] <= x
        if (A[j] <= x) {
            // 5 i = i + 1
            i = i + 1;
            // 6 exchange A[i] with A[j]
            exchange(&A[i], &A[j]);
        }
    }

    // 7 exchange A[i + 1] with A[r]
    exchange(&A[i + 1], &A[r]);

    // 8 return i + 1
    return i + 1;
}

int quicksort(int A[], int p, int r) {
    // 1 if p < r
    if (p < r) {
        // 2 q = PARTITION(A, p, r)
        int q = partition(A, p, r);

        // 3 QUICKSORT(A, p, q - 1)
        quicksort(A, p, q - 1);

        // 4 QUICKSORT(A, q + 1, r)
        quicksort(A, q + 1, r);
    }
    return 0;
}

int main() {
    int n;
    printf("Enter the number of elements: ");
    scanf("%d",&n);

    int A[n];

    // input from the user
    printf("Enter %d integers:\n", n);
    for (int i = 0; i < n; i++) {
            scanf("%d",&A[i]);
    }

    printf("\nOriginal array: ");
     for (int i = 0; i < n; i++) {
            printf("%d ",A[i]);
    }

    // start from 0 index and ends at n-1 index
    quicksort(A, 0, n - 1);

    printf("\nSorted array:   ");
    for (int i = 0; i < n; i++) {
            printf("%d ",A[i]);
    }

    return 0;
}
