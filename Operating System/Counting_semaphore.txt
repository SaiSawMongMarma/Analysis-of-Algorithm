#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

sem_t sem;  // Counting semaphore

void* process(void* arg) {
    int id = *(int*)arg;

    // WAIT(S)
    sem_wait(&sem);

    printf("Process %d: Using resource\n", id);
    sleep(2); // simulate resource use
    printf("Process %d: Releasing resource\n", id);

    // SIGNAL(S)
    sem_post(&sem);
    return NULL;
}

int main() {
    pthread_t t[5];
    int ids[] = {1, 2, 3, 4, 5};

    sem_init(&sem, 0, 3); // Counting semaphore: S = 3 (3 resources)

    for (int i = 0; i < 5; i++)
        pthread_create(&t[i], NULL, process, &ids[i]);

    for (int i = 0; i < 5; i++)
        pthread_join(t[i], NULL);

    sem_destroy(&sem);
    return 0;
}
