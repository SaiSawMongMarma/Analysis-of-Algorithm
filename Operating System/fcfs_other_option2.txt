#include <stdio.h>

int main()
{
    int n, i;
    int AT[10], BT[10], CT[10], TAT[10], WT[10];
    float avgTAT = 0, avgWT = 0;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++)
    {
        printf("Enter AT and BT for P%d: ", i + 1);
        scanf("%d %d", &AT[i], &BT[i]);
    }

    // First process
    CT[0] = AT[0] + BT[0];

    TAT[0] = CT[0] - AT[0];
    WT[0] = TAT[0] - BT[0];

    // Remaining processes
    for(i = 1; i < n; i++)
    {
        if(CT[i - 1] < AT[i])
            CT[i] = AT[i] + BT[i];   // CPU idle
        else
            CT[i] = CT[i - 1] + BT[i];

        TAT[i] = CT[i] - AT[i];
        WT[i] = TAT[i] - BT[i];
    }

    // Calculate averages
    for(i = 0; i < n; i++)
    {
        avgTAT = avgTAT + TAT[i];
        avgWT = avgWT + WT[i];
    }

    avgTAT = avgTAT / n;
    avgWT = avgWT / n;

    printf("\nProcess\tAT\tBT\tCT\tTAT\tWT\n");

    for(i = 0; i < n; i++)
    {
        printf("P%d\t%d\t%d\t%d\t%d\t%d\n",
               i + 1, AT[i], BT[i], CT[i], TAT[i], WT[i]);
    }

    printf("\nAverage TAT = %.2f", avgTAT);
    printf("\nAverage WT  = %.2f\n", avgWT);

    return 0;
}
