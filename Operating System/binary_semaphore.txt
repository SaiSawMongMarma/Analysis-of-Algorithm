#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

sem_t mutex;  // Binary semaphore

void* process(void* arg) {
    int id = *(int*)arg;

    // WAIT(S)
    sem_wait(&mutex); // S = S - 1

    printf("Process %d: Entered critical section\n", id);
    sleep(1); // simulate work
    printf("Process %d: Exiting critical section\n", id);

    // SIGNAL(S)
    sem_post(&mutex); // S = S + 1
    return NULL;
}

int main() {
    pthread_t t1, t2;
    int a = 1, b = 2;

    sem_init(&mutex, 0, 1); // Binary semaphore: S = 1

    pthread_create(&t1, NULL, process, &a);
    pthread_create(&t2, NULL, process, &b);

    pthread_join(t1, NULL);
    pthread_join(t2, NULL);

    sem_destroy(&mutex);
    return 0;
}
