#include <stdio.h>

int main() {
    int n = 6; // Number of processes
    int tq = 2; // Time quantum

    int p[]  = {1, 2, 3, 4,5,6};              // Process IDs
    int at[] = {0, 1, 2, 3, 4, 6};              // Arrival times
    int bt[] = {4, 5, 2, 1, 6, 3};              // Burst times
    int rem_bt[10], ct[10], tat[10], wt[10];
    int time = 0, completed = 0;
    int queue[50];
    int front = 0, rear = 0;
    int visited[10] = {0};
    float avg_tat = 0, avg_wt = 0;

    for (int i = 0; i < n; i++)
        rem_bt[i] = bt[i];

    queue[rear++] = 0; // first process (P1) enters queue
    visited[0] = 1;

    printf("\n=== Round Robin Scheduling ===\n");
    printf("Time Quantum = %d\n", tq);
    printf("\nGantt Chart:\n");

    while (front != rear) {
        int i = queue[front++];
        if (rem_bt[i] > tq) {
            printf("P%d ", p[i]);
            time += tq;
            rem_bt[i] -= tq;
        }
        else {
            printf("P%d ", p[i]);
            time += rem_bt[i];
            rem_bt[i] = 0;
            ct[i] = time;
            completed++;
        }

        // check for newly arrived processes
        for (int j = 0; j < n; j++) {
            if (at[j] <= time && !visited[j] && rem_bt[j] > 0) {
                queue[rear++] = j;
                visited[j] = 1;
            }
        }

        // re-add the current process if not finished
        if (rem_bt[i] > 0)
            queue[rear++] = i;
    }

    printf("\n\nProcess\tAT\tBT\tCT\tTAT\tWT\n");
    for (int i = 0; i < n; i++) {
        tat[i] = ct[i] - at[i];
        wt[i] = tat[i] - bt[i];
        avg_tat += tat[i];
        avg_wt += wt[i];
        printf("P%d\t%d\t%d\t%d\t%d\t%d\n", p[i], at[i], bt[i], ct[i], tat[i], wt[i]);
    }

    avg_tat /= n;
    avg_wt /= n;

    printf("\nAverage Turnaround Time = %.2f", avg_tat);
    printf("\nAverage Waiting Time = %.2f\n", avg_wt);

    return 0;
}
